programa TP2_EJ1
procesos
  proceso juntar (ES f:numero)
  comenzar
    mientras (HayFlorEnLaEsquina)
      tomarFlor
      f:= f+1
  fin
  
  proceso recorrer (ES f:numero)
  comenzar
    repetir 9
      juntar(f)
      mover
    juntar(f)
  fin

areas
  ciudad1: AreaP(1,1,1,10)
  ciudad2: AreaP(2,11,2,20)
robots
  robot juntador1
  variables
    f:numero
    f2:numero
  comenzar
    f:= 0
    recorrer(f)
    EnviarMensaje(f,robot2)
    RecibirMensaje(f2,robot2)
    si (f > f2)
      Informar(f-f2)
  fin
  
  robot juntador2
  variables
    f:numero
    f1:numero
  comenzar
    f:= 0
    recorrer(f)
    EnviarMensaje(f,robot1)
    RecibirMensaje(f1,robot1)
    si (f > f1)
      Informar(f-f1)
  fin
  
variables
  robot1: juntador1
  robot2:juntador2
comenzar
  AsignarArea(robot1,ciudad1)
  AsignarArea(robot2,ciudad2)
  Iniciar(robot1,1,1)
  Iniciar(robot2,2,11)
fin
