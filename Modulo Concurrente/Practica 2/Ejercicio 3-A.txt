programa TP2_EJ3A
procesos
  proceso juntarPapel (ES p:numero)
  comenzar
    mientras (HayPapelEnLaEsquina)
      tomarPapel
      p:= p + 1
  fin
  
  proceso juntarFlor (ES f:numero)
  comenzar
    mientras (HayFlorEnLaEsquina)
      tomarFlor
      f:= f + 1
  fin
  
  proceso recorroFlor (ES f:numero)
  comenzar
    derecha
    repetir 9
      juntarFlor(f)
      mover
    juntarFlor(f)
  fin
  
  proceso recorroPapel (ES p:numero)
  comenzar
    derecha
    repetir 9
      juntarPapel(p)
      mover
    juntarPapel(p)
  fin

areas
  ciudadA1: AreaP(1,1,10,1)
  ciudadA2: AreaP(11,1,20,1)
  ciudadB1: AreaP(1,5,10,5)
  ciudadB2: AreaP(11,5,20,5)
robots
  robot A1
  variables
    p:numero
  comenzar
    p:= 0
    recorroPapel(p)
    EnviarMensaje(p,rA2)  
  fin
  
  robot A2
  variables
    p:numero
  comenzar
    RecibirMensaje(p,rA1)
    recorroPapel(p)
    Informar('Total' , p)
  fin
  
  robot B1 
  variables
    f:numero
  comenzar
    f:=0
    recorroFlor(f)
    EnviarMensaje(f,rB2)
  fin
  
  robot B2
  variables
    f:numero
  comenzar
    RecibirMensaje(f,rB1)
    recorroFlor(f)
    Informar(f)
  fin
  
variables
  rA1: A1
  rA2: A2
  rB1: B1
  rB2: B2
comenzar
  AsignarArea(rA1,ciudadA1)
  AsignarArea(rA2,ciudadA2)
  AsignarArea(rB1,ciudadB1)
  AsignarArea(rB2,ciudadB2)
  Iniciar(rA1,1,1)
  Iniciar(rA2,11,1)
  Iniciar(rB1,1,5)
  Iniciar(rB2,11,5)
fin
