programa TP2_EJ1
procesos
  proceso juntar (ES f:numero)
  comenzar
    mientras (HayFlorEnLaEsquina)
      tomarFlor
      f:= f+1
  fin
  
  proceso recorrer (ES f:numero)
  comenzar
    repetir 9
      juntar(f)
      mover
    juntar(f)
  fin

  proceso maximo (E f:numero; ES max:numero; E quien:numero; ES quienMax:numero)
  comenzar
    si (f > max)
      max:= f
      quienMax:= quien
  fin

areas
  ciudad1: AreaP(1,1,1,10)
  ciudad2: AreaP(2,11,2,20)
  ciudad3: AreaP(3,21,3,30)
  ciudad4: AreaP(4,31,4,40)
  ciudad5: AreaP(5,41,5,50)
  ciudad6: AreaP(6,51,6,60)
  ciudadF: AreaP(2,1,2,1)
  
robots
  robot juntador
  variables
    f,quienSoy:numero
  comenzar
    f:= 0
    RecibirMensaje(quienSoy,fiscal)
    recorrer(f)
    EnviarMensaje(quienSoy,fiscal)
    EnviarMensaje(f,fiscal)
  fin
  
  robot fiscalizador
  variables 
    f,max,quien,quienMax:numero
  comenzar
    max:=0
    EnviarMensaje(1,robot1)
    EnviarMensaje(2,robot2)
    EnviarMensaje(3,robot3)
    EnviarMensaje(4,robot4)
    EnviarMensaje(5,robot5)
    EnviarMensaje(6,robot6)
    repetir 6
      RecibirMensaje(quien,*)
      si (quien = 1)
        RecibirMensaje(f,robot1)
      sino
        si (quien = 2)
          RecibirMensaje(f,robot2)
        sino
          si (quien = 3)
            RecibirMensaje(f,robot3)
          sino
            si (quien = 4)
              RecibirMensaje(f,robot4)
            sino
              si (quien = 5)
                RecibirMensaje(f,robot5)
              sino
                RecibirMensaje(f,robot6)           
      maximo(f,max,quien,quienMax)  
    Informar(quienMax, max)
  fin
  
variables
  robot1: juntador
  robot2: juntador
  robot3: juntador
  robot4: juntador
  robot5: juntador
  robot6: juntador
  fiscal:fiscalizador
comenzar
  AsignarArea(robot1,ciudad1)

  AsignarArea(robot3,ciudad3)

  AsignarArea(robot5,ciudad5)
  AsignarArea(robot6,ciudad6)
  AsignarArea(fiscal,ciudadF)

  Iniciar(robot3,3,21)
  Iniciar(robot4,4,31)
  Iniciar(robot5,5,41)
  Iniciar(robot6,6,51)
  Iniciar(fiscal,2,1)
fin
