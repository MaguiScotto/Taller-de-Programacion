programa turnoB
procesos
  proceso juntar (ES f:numero)
  comenzar
    mientras HayFlorEnLaEsquina
      tomarFlor
      f:=f+1
  fin
  
  proceso definir 
  variables
    intentos:numero
  comenzar
    Random(intentos,3,6)
    EnviarMensaje(intentos,r1)
    Random(intentos,3,6)
    EnviarMensaje(intentos,r2)
    Random(intentos,3,6)
    EnviarMensaje(intentos,r3)
  fin
  
  proceso identificar
  comenzar
    EnviarMensaje(1,r1)
    EnviarMensaje(2,r2)
    EnviarMensaje(3,r3)
  fin
  
  proceso depositar(E ave:numero; E cal:numero)
  comenzar
    BloquearEsquina(5,5)
    Pos(5,5)
    si HayFlorEnLaBolsa
      depositarFlor
    Pos(ave,cal)
    LiberarEsquina(5,5)
  fin
 
areas
  cuadrante: AreaPC(10,10,30,30)
  esquina: AreaC (5,5,5,5)
  esqFiscal: AreaP (9,9,9,9)
  esqRecol: AreaPC(1,1,3,1)
robots
  robot recolector
  variables
    quienSoy,intentos,f,iniAv,iniCa,ave,cal,total:numero
  comenzar
    f:=0
    total:= f
    iniAv:=PosAv
    iniCa:=PosCa
    RecibirMensaje(quienSoy,fiscal)
    RecibirMensaje(intentos,fiscal)
    repetir intentos
      Random(ave,10,30)
      Random(cal,10,30)
      BloquearEsquina(ave,cal)
      Pos(ave,cal)
      juntar(f)
      total:= total + f
      repetir f
        depositar(ave,cal)
      Pos(iniAv,iniCa)
      LiberarEsquina(ave,cal)
    EnviarMensaje(quienSoy,fiscal)
    EnviarMensaje(total,fiscal) 
  fin 
  
  robot fiscalizador
  variables
    f,quien,max,maxQuien:numero
  comenzar
    max:=0
    identificar
    definir
    repetir 3
      RecibirMensaje(quien,*)
      si (quien=1)
        RecibirMensaje(f,r1)
      sino
        si (quien=2)
          RecibirMensaje(f,r2)
        sino
          RecibirMensaje(f,r3)
      si (f > max)
        max:= f
        maxQuien:= quien  
    Pos(5,5)
    mientras HayFlorEnLaEsquina
      tomarFlor
    Informar(quien)    
  fin
  
variables
  r1,r2,r3:recolector
  fiscal:fiscalizador
comenzar
  AsignarArea(r1,esqRecol)
  AsignarArea(r1,cuadrante)
  AsignarArea(r1,esquina)
  AsignarArea(r2,esqRecol)
  AsignarArea(r2,cuadrante)
  AsignarArea(r2,esquina)
  AsignarArea(r3,esqRecol)
  AsignarArea(r3,cuadrante)
  AsignarArea(r3,esquina)
  AsignarArea(fiscal,esqFiscal)
  AsignarArea(fiscal,esquina)
  Iniciar(r1,1,1)
  Iniciar(r2,2,1)
  Iniciar(r3,3,1)
  Iniciar(fiscal,9,9)
fin
