programa TP1_EJ1
procesos 
  proceso contar (ES no:numero; ES f:numero)
  comenzar
    si (HayFlorEnLaEsquina)
      mientras (HayFlorEnLaEsquina)
        tomarFlor
        f:= f + 1
    sino 
      no:= no + 1
  fin 
  
  proceso avanzar (ES no:numero; ES f:numero) 
  comenzar
    repetir 99
      contar (no,f)
      mover
    contar (no,f)
    repetir f
      si (HayFlorEnLaBolsa)
        depositarFlor
    Informar(PosAv, no)
    Informar(PosAv, f)
  fin
