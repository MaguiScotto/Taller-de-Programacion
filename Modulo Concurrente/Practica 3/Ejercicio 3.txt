programa TP3_EJ3
procesos
  proceso identificar 
  comenzar
    EnviarMensaje(1,r1)
    EnviarMensaje(2,r2)
    EnviarMensaje(3,r3)
    EnviarMensaje(4,r4)
  fin
  
  proceso enviarEsquina (E av:numero; E ca:numero)
  comenzar
    EnviarMensaje(av,r1)
    EnviarMensaje(ca,r1)
    EnviarMensaje(av,r2)
    EnviarMensaje(ca,r2)
    EnviarMensaje(av,r3)
    EnviarMensaje(ca,r3)
    EnviarMensaje(av,r4)    
    EnviarMensaje(ca,r4)
  fin
  
  proceso juntar (ES f:numero; ES seguir:boolean)
  comenzar
    si(HayFlorEnLaEsquina)
      tomarFlor
      f:=f+1
    sino
      seguir:=F
  fin
  
areas
  cuadrante: AreaPC (2,2,10,10)
  esq: AreaC (1,1,5,1)
robots
  robot jefe
  variables
    av,ca:numero
    quien,cant:numero
    max,quienMax:numero
  comenzar
    max:= -1
    identificar
    Random(av,2,10)
    Random(ca,2,10)
    enviarEsquina(av,ca)
    repetir 4
      RecibirMensaje(quien,*)
      si(quien=1)
        RecibirMensaje(cant,r1)
      sino
        si (quien=2)
          RecibirMensaje(cant,r2)
        sino
          si (quien=3)
            RecibirMensaje(cant,r3)
          sino
            RecibirMensaje(cant,r4)
      si(cant>max)
        max:=cant
        quienMax:=quien
    Informar(quienMax)
  fin  
    
  robot juntador
  variables
    quienSoy,av,ca,f,avIni,caIni:numero
    seguir:boolean
  comenzar
    f:= 0
    avIni:= PosAv
    caIni:= PosCa
    seguir:= V
    RecibirMensaje(quienSoy,jefazo)
    RecibirMensaje(av,jefazo)
    RecibirMensaje(ca,jefazo)
    mientras seguir
      BloquearEsquina(av,ca)
      Pos(av,ca)
      juntar(f,seguir)
      Pos(avIni,caIni)
      LiberarEsquina(av,ca)
    EnviarMensaje(quienSoy,jefazo)
    EnviarMensaje(f,jefazo)
  fin
    
variables
  r1:juntador
  r2:juntador
  r3:juntador
  r4:juntador
  jefazo:jefe
comenzar
  AsignarArea(r1,cuadrante)
  AsignarArea(r2,cuadrante)
  AsignarArea(r3,cuadrante)
  AsignarArea(r4,cuadrante)
  AsignarArea(r1,esq)
  AsignarArea(r2,esq)
  AsignarArea(r3,esq)
  AsignarArea(r4,esq)
  AsignarArea(jefazo,esq)
  Iniciar(jefazo,1,1)
  Iniciar(r1,2,1)
  Iniciar(r2,3,1)
  Iniciar(r3,4,1)
  Iniciar(r4,5,1)
fin
